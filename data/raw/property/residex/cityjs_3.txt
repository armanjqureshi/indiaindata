$(document).ready(function () {
    google.charts.load('current', { packages: ['corechart', 'line'] });
    $('#div_Dll_Unit').hide();
    $('#hideContentWhileTableShow').show();

    /* ------------------ Defaults ------------------ */
    function ensureDefaults() {
        // HousingPriceIndices
        let $hpiRadios = $('input:radio[name=HousingPriceIndices]');
        if (!$hpiRadios.is(':checked')) {
            $hpiRadios.filter('[value=HFC]').prop('checked', true);
        }
        $("#selectedHousingPriceIndices").val($('input[name=HousingPriceIndices]:checked').val());

        // CitywiseHPI
        let $cityRadios = $('input:radio[name=CitywiseHPI]');
        if (!$cityRadios.is(':checked')) {
            $cityRadios.filter('[value=1]').prop('checked', true);
        }
        $("#selectedCitywiseHPI").val($('input[name=CitywiseHPI]:checked').val());

        // UnitSize
        let $unitCheckboxes = $('input:checkbox[name=UnitSizeinCarpetArea]');
        if (!$unitCheckboxes.is(':checked')) {
            $unitCheckboxes.filter('[value=5]').prop('checked', true);
        }
        $("#chkUnitSizeinCarpetArea").val($('input[name=UnitSizeinCarpetArea]:checked').val());

        // Visibility rules
        toggleUnitSizeVisibility();
    }

    function toggleUnitSizeVisibility() {
        const hpiValue = $("#selectedHousingPriceIndices").val();
        const citywiseValue = $("#selectedCitywiseHPI").val();

        if (hpiValue === "BOTH_INDEX") {
            $("#chkbefore1718Div").hide();
            $("#UnitSizeinCarpetArea").hide();
            $('input:checkbox[name=UnitSizeinCarpetArea]').prop('checked', false)
                .filter('[value=5]').prop('checked', true);
            $("#chkUnitSizeinCarpetArea").val("5");
        } else {
            $("#chkbefore1718Div").show();
            if (citywiseValue === "2") {
                $("#UnitSizeinCarpetArea").show();
            } else {
                $("#UnitSizeinCarpetArea").hide();
            }
        }
    }

    function refreshCharts() {
        ensureDefaults(); // always set hidden fields first
        google.charts.setOnLoadCallback(showChart);
        google.charts.setOnLoadCallback(showChartCurrentQtr);
    }

    function initializeDashboard() {
        ensureDefaults();

        const cityId = $("#Cityid").val();

        if (cityId && cityId !== "0") {
            BindStateBasedOnCity(cityId).then(stateId => {
                return BindCityDropDownList("ddlCity", stateId);
            }).then($cityDdl => {
                $cityDdl.val(cityId);
                $("#selectedCity").val(cityId);
                refreshCharts();
            }).catch(err => {
                console.error("Initialization failed:", err);
            });
        } else {
            BindStateDropDownList("ddlState").then($stateDdl => {
                const defaultStateIndex = 11;
                $stateDdl.prop('selectedIndex', defaultStateIndex).trigger('change');
                const selectedState = $stateDdl.val();
                return BindCityDropDownList("ddlCity", selectedState);
            }).then($cityDdl => {
                $cityDdl.prop('selectedIndex', 0);
                $("#selectedCity").val($cityDdl.val());
                refreshCharts();
            });
        }
    }

    initializeDashboard();

    /* ------------------ Event Bindings ------------------ */
    $("#ddlState").change(function () {
        const selectedState = $(this).val();
        $("#selectedState").val(selectedState);

        if (selectedState > 0) {
            BindCityDropDownList("ddlCity", selectedState).then($cityDdl => {
                $cityDdl.prop('selectedIndex', 0);
                $("#selectedCity").val($cityDdl.val());
                refreshCharts();
            });
        } else {
            $("#ddlCity").html('<option value="">Please select a valid state to view the data.</option>');
            $("#selectedCity").val('');
        }
    });

    //Fixed
    // $("#ddlCity").on("change", function () {

    $("#ddlCity").change(function () {
        const selectedState = $("#selectedState").val();
        if (!selectedState) {
            alert("Please select a state first.");
            return;
        }
        $("#selectedCity").val($(this).val());
        refreshCharts();
    });

    // Radio handlers
    $('input[type=radio][name=HousingPriceIndices]').change(function () {
        $("#selectedHousingPriceIndices").val(this.value);
        toggleUnitSizeVisibility();
        refreshCharts();
    });

    $('input[type=radio][name=CitywiseHPI]').change(function () {
        $("#selectedCitywiseHPI").val(this.value);
        toggleUnitSizeVisibility();
        refreshCharts();
    });

    // UnitSize checkboxes (single select + default enforcement)
    $('input[type=checkbox][name=UnitSizeinCarpetArea]').change(function () {
        const checkboxes = $('input[type=checkbox][name=UnitSizeinCarpetArea]');
        const checkedBoxes = checkboxes.filter(':checked');

        if (this.checked) {
            checkboxes.not(this).prop('checked', false); // enforce single selection
            $("#chkUnitSizeinCarpetArea").val(this.value);
        } else {
            $("#chkUnitSizeinCarpetArea").val('');
        }

        if (checkedBoxes.length === 0) {
            const defaultCheckbox = checkboxes.filter('[value=9]');
            if (defaultCheckbox.length) {
                defaultCheckbox.prop('checked', true);
                $("#chkUnitSizeinCarpetArea").val("9");
                alert("At least one checkbox must be selected.");
            }
        }

        refreshCharts();
    });

    // chkbefore1718 checkbox
    $('input[type=checkbox][name=chkbefore1718]').change(function () {
        $("#chkbefore1718Checked").val(this.checked ? this.value : '');

        const $radios = $('input:checkbox[name=UnitSizeinCarpetArea]');
        if ($radios.is(':checked') === false) {
            $radios.filter('[value=5]').prop('checked', true);
            $("#chkUnitSizeinCarpetArea").val("5");
        }

        refreshCharts();
    });

    // Unit dropdown
    $("#ddlUnit").change(function () {
        const selectedUnit = $(this).find('option:selected').text().split('.Per ')[1];
        $("#selectedUnit").val(selectedUnit);
        showTable(selectedUnit);
        showTableCurrentQtr(selectedUnit);
    });

    // Download
    $("#btnDownload").click(function () {
        const city = $('#ddlCity option:selected').text();
        $("#tblPriceTrend").click();
        setTimeout(() => {
            tableToExcel("divPriceTrend", `${city}-City HPI Data Previous-Q (Base Year 2024)`);
            tableToExcel("divPriceTrend1", `${city}-City HPI Data Current-Q (Base Year 2024)`);
        }, 500);
    });
});

function FileDownload1(downloadType) {
    var c = '0';
    if ($('#dwncityname option').length === $('#dwncityname').val().length || $('#dwncityname').val().length === 0) {
        c = '0';
    } else {
        c = $('#dwncityname').val().join();
    }

    var q = '0';
    if ($('#dwnqtrname option').length === $('#dwnqtrname').val().length || $('#dwnqtrname').val().length === 0) {
        q = '0';
    } else {
        q = $('#dwnqtrname').val().join();
    }

    var pricemtdid = 2; // Example value, set it as needed
    var lang = 0; // Assuming language is 0

    var DWN = $('#rdoDWN').val();
    var IA = $('#rdodwnIA').val();

    var reporttype = "PIVOT_" + DWN + "_" + IA;

    $.ajax({
        url: "/DashboardBY2024/GetFileDataToDownload", // Ensure this URL points to your actual controller endpoint
        type: "POST",
        data: {
            reporttype: reporttype,
            qtr: q,
            cityid: c,
            pricemtdid: pricemtdid,
            lang: lang
        },
        success: function (response) {
            if (response.status === "Success") {
                var arr = JSON.parse(response.data);

                if (arr.length > 0) {
                    var headers = Object.keys(arr[0]);

                    var tableHtml = '<table border="1">';
                    tableHtml += '<tr>';
                    headers.forEach(function (header) {
                        tableHtml += '<th>' + header + '</th>';
                    });
                    tableHtml += '</tr>';

                    arr.forEach(function (item) {
                        tableHtml += '<tr>';
                        headers.forEach(function (header) {
                            tableHtml += '<td>' + item[header] + '</td>';
                        });
                        tableHtml += '</tr>';
                    });

                    tableHtml += '</table>';
                    document.body.insertAdjacentHTML('beforeend', '<div id="exportTable" style="display:none;">' + tableHtml + '</div>');

                    tableToExcel('exportTable', 'Residex_Data');
                } else {
                    alert('No data available.');
                }
            } else {
                alert('Error: ' + response.Message);
            }
        },
        error: function () {
            alert('Error fetching data.');
        }
    });
}


function tblnamechange() {
    var btn = document.getElementById('tblPriceTrend');
    var lang = $('#LngId').val();
    if (btn.value === (lang === '0' ? 'View Table' : 'तालिका')) {
        btn.value = (lang === '0' ? 'View Chart' : 'चार्ट');
        showTable(selectedUnit); showTableCurrentQtr(selectedUnit);

        setTimeout(function () {
            $("#ddlUnit").val($("#ddlUnit option:first").val()).trigger("change");
        }, 200);

        $('#hideContentWhileTableShow').hide();
        $('#div_Dll_Unit').show();
        
    } else {
        btn.value = (lang === '0' ? 'View Table' : 'तालिका');
        showChart(); showChartCurrentQtr();
        $('#hideContentWhileTableShow').show();
        $('#div_Dll_Unit').hide();
    }
}


function fetchData(callback) {
    var selectedquarter = 2;
    var chkbefore1718 = $("#chkbefore1718Checked").val();
    var selectedCity = $("#selectedCity").val();
    var lookup = $("#selectedHousingPriceIndices").val();

    $.ajax({
        url: "/DashboardBY2024/Get_NHBResiCersai_PriceTrend",
        type: "POST",
        data: {
            reporttype: "CITY",
            lookup: lookup,
            cityid: selectedCity,
            pincode: 0,
            wardid: 0,
            pricemtdid: selectedquarter,
            locid: "0",
            chkbefore1718: chkbefore1718
        },
        success: function (response) {
            if (response.status === "Success" && response.data) {
                const jsonData = JSON.parse(response.data); // Parse JSON string 
                callback(jsonData);
            } else {
                console.error(response.message || "No data available.");

                ShowErrorMessage(response.message || "No data available.");
            }
        },
        error: function (xhr, status, error) {
            console.error("AJAX Error:", status, error);
            ShowErrorMessage("Error while fetching data.");
        }
    });

}

function showTable(unit) {
    fetchData((jsonData) => {

        tableHtml += `</tbody></table>`;
        $("#divPriceTrend").html(tableHtml);
    });

function generateTableRow(hpiType, row, unit) {

    }

    if (hpiType === "BOTH_INDEX") {
        return ` <tr> <td>${row['QTR_YEAR']}</td> <td>${row['HFC_INDEX'] || "N/A"}</td> <td>${row[`HFC_WTAVG_COMPOSITE${unitSuffix}`] || "N/A"}</td> <td>${row['UC_INDEX'] || "N/A"}</td> <td>${row[`UC_WTAVG_COMPOSITE${unitSuffix}`] || "N/A"}</td> </tr>`;
    }
    return '';
}

function ShowErrorMessage(message) {

function fetchDataCurrentQtr(callback) {

    var selectedquarter = 1;
    var chkbefore1718 = $("#chkbefore1718Checked").val();
    var selectedCity = $("#selectedCity").val();
    var lookup = $("#selectedHousingPriceIndices").val();

    $.ajax({
        url: "/DashboardBY2024/Get_NHBResiCersai_PriceTrend",
        type: "POST",
        data: {
            reporttype: "CITY",
            lookup: lookup,
            cityid: selectedCity,
            pincode: 0,
            wardid: 0,
            pricemtdid: selectedquarter,
            locid: "0",
            chkbefore1718: chkbefore1718
        },
        success: function (response) {
            if (response.status === "Success" && response.data) {
                const jsonData = JSON.parse(response.data); // Parse JSON string 
                callback(jsonData);
            } else {
                console.error(response.message || "No data available.");

                ShowErrorMessage(response.message || "No data available.");
            }
        },
        error: function (xhr, status, error) {
            console.error("AJAX Error:", status, error);
            ShowErrorMessage("Error while fetching data.");
        }
    });
}

            if (rbHousingPriceIndices == "HFC") {
                chartViewBasedOnRB.hideColumns([2]);
            }
            else if (rbHousingPriceIndices == "UC") {
                var clr = ['#dc3912'];
                options.colors = clr;
                chartViewBasedOnRB.hideColumns([1]);
            }

            if ($('#chkbefore1718').is(':checked')) {
                $('#liCompositePrice').hide();
                if ($('input[type=checkbox][name=UnitSizeinCarpetArea]:checked').not("#CompositePrice").length === 0) {
                    $('#60sqmt').prop('checked', true);
                }
                $('#CompositePrice').prop('checked', false);
            } else {
                $('#liCompositePrice').show();
            }

            chart.draw(chartViewBasedOnRB, options);

        });
