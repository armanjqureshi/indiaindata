function FileDownload(downloadType) {

    var cityValues = $('#dwncityname').val() || [];
    var quarterValues = $('#dwnqtrname').val() || [];

    var c = (cityValues.length === $('#dwncityname option').length || cityValues.length === 0)
        ? '0'
        : cityValues.join();

    var q = (quarterValues.length === $('#dwnqtrname option').length || quarterValues.length === 0)
        ? '0'
        : quarterValues.join();

    var pricemtdid = 2; 
    var lang = 0;      

    var DWN = $('#rdoDWN').val();
    var IA = $('#rdodwnIA').val();

    var reporttype = "PIVOT_" + DWN + "_" + IA;

    $.ajax({
        url: "/Dashboard/GetFileDataToDownload", 
        type: "POST",
        data: {
            reporttype: reporttype,
            qtr: q,
            cityid: c,
            pricemtdid: pricemtdid,
            lang: lang
        },
        success: function (response) {
            if (response.status === "Success") {
                var arr = JSON.parse(response.data);

                if (arr.length > 0) {
                    var headers = Object.keys(arr[0]);

                    var tableHtml = '<table border="1">';
                    tableHtml += '<tr>';
                    headers.forEach(function (header) {
                        tableHtml += '<th>' + header + '</th>';
                    });
                    tableHtml += '</tr>';

                    arr.forEach(function (item) {
                        tableHtml += '<tr>';
                        headers.forEach(function (header) {
                            tableHtml += '<td>' + item[header] + '</td>';
                        });
                        tableHtml += '</tr>';
                    });

                    tableHtml += '</table>';

                    $('#exportTable').remove();

                    document.body.insertAdjacentHTML('beforeend', '<div id="exportTable" style="display:none;">' + tableHtml + '</div>');

                    tableToExcel('exportTable', 'Residex_Data');
                } else {
                    alert('No data available.');
                }
            } else {
                alert('Error: ' + response.Message);
            }
        },
        error: function () {
            alert('Error fetching data.');
        }
    });
}

function fnCircle1(cityid) {
  
    fetch(`/DashboardBY2018/EncryptCityId?cityid=${cityid}`).then(response => response.text()).then(encryptedCityId => {
        var url = `/DashboardBY2018/DashboardBY2018?sfjy=${encodeURIComponent(encryptedCityId)}`;

function getcityprices() {
    var reporttype = $("#hdnPrices_yoy").val() || "CITY_HFC";
    var qtr = $("#selectedQuarterYoY").val();
    //var qtr = $("#selectedQuarterYoY").val() || "25-26 Q3";

    $.ajax({
        url: "Dashboard/GetResidexPriceScroll",
        type: "POST",
        data: { reporttype: reporttype, qtr: qtr, cityid: "0", pricemtdid: 2 },
        success: function (response) {
            arr = [];
            if (response.status === "Success") {
                // Parse response and populate city data 
                var jsondata = JSON.parse(response.data);
                var cities = '';
                for (var i = 0; i < jsondata.length; i++) {
                    arr.push({
                        stateid: jsondata[i]["STATE_ID"],
                        cityid: jsondata[i]["CITY_ID"],
                        city: ($('#LngId').val() == '0' ? jsondata[i]["CITY"] : jsondata[i]["CITY_NAME_HINDI"]),
                        yoy: jsondata[i]["INDEX_YOY"],
                        qoq: jsondata[i]["INDEX_QOQ"],
                        ytd: jsondata[i]["INDEX_YTD"]
                    });

                    cities += '<option value="' + jsondata[i]["CITY_ID"] + '">' +
                        ($('#LngId').val() == '0' ? jsondata[i]["CITY"] : jsondata[i]["CITY_NAME_HINDI"]) +
                        '</option>';
                }
                $('#dwncityname1').empty().append(cities);
                setscrolldata();
            }

            d3.selectAll("path[class^='state_']").remove();
            d3.selectAll("circle[class^='state_']").remove();

            arr.forEach(item => {
                    renderCityData(item);
            });
        },
        error: function () {
            alert('Error fetching data.');
        }
    });
}

function renderCityData(item) {
    var c = ccoord[0][item.cityid];
    var rsizeI = 4; // Small radius
    var rsize = 7;  // Large radius
    var csizeI = 3; // Small circle size
    var csize = 6;  // Large circle size

    var gstate = d3.select("#gstate_" + item.stateid);
    var gcstate = d3.select("#gcstate_" + item.stateid);

    if (item.qoq > 0.5) { // Up arrow for positive QoQ
        var pd = "M" + c.x + " " + c.y + " L" + (c.x + rsizeI) + " " + (c.y + rsizeI) + " L" + (c.x - rsizeI) + " " + (c.y + rsizeI) + " Z";
        var pd2 = "M" + c.sx + " " + c.sy + " L" + (c.sx + rsize) + " " + (c.sy + rsize) + " L" + (c.sx - rsize) + " " + (c.sy + rsize) + " Z";

        gstate.append("path")
            .attr("stroke", "transparent")
            .attr("stroke-width", rsizeI)
            .attr("class", "state_" + item.stateid)
            .attr("d", pd)
            .attr("fill", c.color)
            .attr("onclick", `fnCircle(${item.cityid});`)
            //.attr("onmouseover", `fnCircleHover('${item.city} (${item.qoq}%)');`)
            //.attr("onmouseout", "fnCircleOut();")
            .append("title").html(item.city + ' (' + item.yoy + '%)');

        gcstate.append("path")
            .attr("stroke", "transparent")
            .attr("stroke-width", rsize)
            .attr("class", "state_" + item.stateid)
            .attr("d", pd2)
            .attr("fill", c.color)
            .attr("onclick", `fnCircle(${item.cityid});`)
            //.attr("onmouseover", `fnCircleHover('${item.city} (${item.qoq}%)');`)
            //.attr("onmouseout", "fnCircleOut();")
            .append("title").html(item.city + ' (' + item.yoy + '%)');
    } else if (item.qoq < -0.5) { // Down arrow for negative QoQ
        var pd = "M" + c.x + " " + c.y + " L" + (c.x + rsizeI) + " " + (c.y - rsizeI) + " L" + (c.x - rsizeI) + " " + (c.y - rsizeI) + " Z";
        var pd2 = "M" + c.sx + " " + c.sy + " L" + (c.sx + rsize) + " " + (c.sy - rsize) + " L" + (c.sx - rsize) + " " + (c.sy - rsize) + " Z";

        gstate.append("path")
            .attr("stroke", "transparent")
            .attr("stroke-width", rsizeI)
            .attr("class", "state_" + item.stateid)
            .attr("d", pd)
            .attr("fill", c.color)
            .attr("onclick", `fnCircle(${item.cityid});`)
            //.attr("onmouseover", `fnCircleHover('${item.city} (${item.qoq}%)');`)
            //.attr("onmouseout", "fnCircleOut();")
            .append("title").html(item.city + ' (' + item.yoy + '%)');

        gcstate.append("path")
            .attr("stroke", "transparent")
            .attr("stroke-width", rsize)
            .attr("class", "state_" + item.stateid)
            .attr("d", pd2)
            .attr("fill", c.color)
            .attr("onclick", `fnCircle(${item.cityid});`)
            //.attr("onmouseover", `fnCircleHover('${item.city} (${item.qoq}%)');`)
            //.attr("onmouseout", "fnCircleOut();")
            .append("title").html(item.city + ' (' + item.yoy + '%)');
    } else { // Circle for neutral QoQ
        gstate.append("circle")
            .attr("stroke", "transparent")
            .attr("stroke-width", rsizeI)
            .attr("id", "cty_" + item.cityid)
            .attr("class", "state_" + item.cityid)
            .attr("cx", c.x)
            .attr("cy", c.y)
            .attr("r", csizeI)
            .attr("fill", c.color)
            .attr("data-name", item.city)
            .attr("onclick", `fnCircle(${item.cityid});`)
            //.attr("onmouseover", `fnCircleHover('${item.city} (${item.qoq}%)', '${item.cityid}');`)
            //.attr("onmouseout", "fnCircleOut();")
            .append("title").html(item.city + ' (' + item.yoy + '%)');

        gcstate.append("circle")
            .attr("stroke", "transparent")
            .attr("stroke-width", rsize)
            .attr("class", "state_" + item.cityid)
            .attr("cx", c.sx)
            .attr("cy", c.sy)
            .attr("r", csize)
            .attr("fill", c.color)
            .attr("data-name", item.city)
            .attr("onclick", `fnCircle(${item.cityid});`)
            //.attr("onmouseover", `fnCircleHover('${item.city} (${item.qoq}%)');`)
            //.attr("onmouseout", "fnCircleOut();")
            .append("title").html(item.city + ' (' + item.yoy + '%)');
    }
}

function loadCityChart() {
    var lookup = $("#hdnPrices").val();
    if (!lookup) {
        lookup = "HFC";
        $("#hdnPrices").val(lookup);
    }
    var state = $("#selectedState").val() || "0";
    $.ajax({
        url: 'Dashboard/Get_NHBResiCersai_PriceTrend_ForCity',
        type: 'POST',
        data: { reporttype: "ALLCITY", lookup: lookup, cityid: "0", pincode: "0", wardid: "0", pricemtdid: "2", locid: "0", stateid: state, chkbefore1718Citywise: $("#chkbefore1718CitywiseChecked").val() },
        success: function (response) {
            if (response.status === "Success") {
                allCityData = JSON.parse(response.data);
                cityList = Object.keys(allCityData[0]).filter(key => key !== 'QTR_YEAR');

                renderChart();
            } else {
                document.getElementById('chart_allcity').innerHTML = "No data available";
            }
        }, error: function () {
            alert('Error fetching data.');
        }
    });
}

function tblnamechange() {
    var btn = document.getElementById('tblPriceTrend');
    var lang = $('#LngId').val();
    if (btn.value === (lang === '0' ? 'View Table' : 'तालिका')) {
        btn.value = (lang === '0' ? 'View Chart' : 'चार्ट'); showTable();
    } else {
        btn.value = (lang === '0' ? 'View Table' : 'तालिका'); showChart();
    }
}

function fetchData(callback) {
    var selectedquarter = $("#selectedQuarter").val();
    //var rbHousingPriceIndices = $("#selectedHousingPriceIndices").val();
    var chkbefore1718 = $("#chkbefore1718Checked").val();

    $.ajax({
        url: "Dashboard/Get_NHBResiCersai_PriceTrend",
        type: "POST",
        data: {
            reporttype: "COUNTRY",
            lookup: "BOTH_INDEX",
            cityid: 0, pincode: 0, wardid: 0,
            pricemtdid: selectedquarter, locid: 0,
            chkbefore1718: chkbefore1718
        },
        success: function (response) {
            if (response.status === "Success" && response.data) {
                const jsonData = JSON.parse(response.data);
                callback(jsonData);
            }
            else {

                ShowErrorMessage(response.Message || "No data available");
            }
        }, error: function () {
            ShowErrorMessage("Error while fetching data.");
        }
    });
}

function showTable() {
    fetchData(function (jsonData) {
        let tableHtml = ` <table style="width: 100%; text-align: center; border-collapse: collapse;" class="table table-bordered table-responsive-sm table-striped"> <thead class="table-secondary" style="position: sticky;top: -1px;"  > <tr> <th>Quarter</th> <th>HPI@Assessment Prices</th> <th>HPI@Market Prices</th> </tr> </thead> <tbody>`;
        jsonData.forEach((row) => {
            tableHtml += ` <tr> <td>${row['QTR_YEAR']}</td>
            <td>${row['HFC_COUNTRY_INDEX'] || "N/A"}</td>
            <td>${row['UC_COUNTRY_INDEX'] || "N/A"}</td> </tr>`;
        }); tableHtml += `</tbody></table>`;
        $("#divPriceTrend").html(tableHtml);
    });
}

function showChart1() {
    fetchData(function (jsonData) {
        const dataTable = new google.visualization.DataTable();
        dataTable.addColumn('string', 'Quarter');
        dataTable.addColumn('number', 'HPI@Assessment Prices');
        dataTable.addColumn('number', 'HPI@Market Prices');
        jsonData.forEach((row) => {
            dataTable.addRow([row['QTR_YEAR'],
            parseFloat(row['HFC_COUNTRY_INDEX']) || null,
            parseFloat(row['UC_COUNTRY_INDEX']) || null]);
        });
        options = {
            width: '100%', height: 350, pointSize: 5,
            legend: { position: 'top' },
            hAxis: { slantedText: true, slantedTextAngle: 45 },
            vAxis: { viewWindow: { min: 0 } },
            chartArea: { left: 50, width: '100%', height: '60%' },
            title: 'Price Trend Analysis', interpolateNulls: true
        };
        chart = new google.visualization.LineChart(document.getElementById('divPriceTrend'));
        chartViewBasedOnRB = new google.visualization.DataView(dataTable);
        var rbHousingPriceIndices = $("#selectedHousingPriceIndices").val();

        if (rbHousingPriceIndices == "HFC") {
            chartViewBasedOnRB.hideColumns([2]);
        }
        else if (rbHousingPriceIndices == "UC") {
            var clr = ['#dc3912'];
            options.colors = clr;
            chartViewBasedOnRB.hideColumns([1]);
        }
        chart.draw(chartViewBasedOnRB, options);

    });
}
